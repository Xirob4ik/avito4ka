cmake_minimum_required(VERSION 3.22)

# Static MSVC runtime: the .vst3 does not require the Visual C++ Redistributable on the user's PC.
# Must be set before project().
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING "")

# macOS: one universal binary for Apple Silicon and Intel Macs, running on macOS 10.13 and later.
# Must be set before project().
set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "macOS architectures")
set(CMAKE_OSX_DEPLOYMENT_TARGET "10.13" CACHE STRING "Minimum macOS version")

project(Clipline VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set_property(GLOBAL PROPERTY USE_FOLDERS ON)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

# ------------------------------------------------------------------------------------------------
# Options
# ------------------------------------------------------------------------------------------------
option(CLIPLINE_BUILD_TESTS        "Build DSP and plugin tests"                              ON)
option(CLIPLINE_BUILD_STANDALONE   "Also build a Standalone app (needs audio device headers)" OFF)
option(CLIPLINE_COPY_AFTER_BUILD   "Copy the VST3 into the system VST3 folder after build"   OFF)
option(CLIPLINE_ENABLE_LTO         "Link-time optimisation for the plugin target"            ON)
set(CLIPLINE_JUCE_DIR "" CACHE PATH "Path to a local JUCE checkout (otherwise fetched from GitHub)")

# ------------------------------------------------------------------------------------------------
# Branding. Everything user-visible that should be replaced for a release lives here and in
# Source/Branding.h (see docs/BRANDING.md).
# ------------------------------------------------------------------------------------------------
set(CLIPLINE_PRODUCT_NAME  "Clipline")
set(CLIPLINE_COMPANY_NAME  "Clipline Audio")
set(CLIPLINE_MANUFACTURER_CODE "Clpl")   # exactly 4 chars, at least one upper-case letter
set(CLIPLINE_PLUGIN_CODE       "Clp1")   # exactly 4 chars, unique per product
set(CLIPLINE_BUNDLE_ID     "com.cliplineaudio.clipline")

include(cmake/FetchJuce.cmake)

# ------------------------------------------------------------------------------------------------
# Source lists. Each target compiles the sources it needs: JUCE modules are INTERFACE libraries whose
# implementation files are built into every consuming target, so sharing code through an intermediate
# static library that also links modules would duplicate symbols.
# ------------------------------------------------------------------------------------------------
# DSP core: depends only on juce_core / juce_audio_basics / juce_dsp (no GUI, no plugin client).
file(GLOB CLIPLINE_DSP_SOURCES CONFIGURE_DEPENDS
    Source/DSP/*.cpp Source/DSP/*.h
    Source/Presets/FactoryPresets.cpp Source/Presets/FactoryPresets.h
    Source/ParameterIds.h)

# Everything that makes up the plugin except the format entry point (PluginEntry.cpp, which
# defines createPluginFilter() and is only compiled into the actual plugin target).
file(GLOB CLIPLINE_TOP_SOURCES CONFIGURE_DEPENDS Source/*.cpp Source/*.h)
list(REMOVE_ITEM CLIPLINE_TOP_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/Source/PluginEntry.cpp")
file(GLOB CLIPLINE_UI_SOURCES      CONFIGURE_DEPENDS Source/UI/*.cpp Source/UI/*.h)
file(GLOB CLIPLINE_PRESET_SOURCES  CONFIGURE_DEPENDS Source/Presets/*.cpp Source/Presets/*.h)
set(CLIPLINE_PLUGIN_SOURCES
    ${CLIPLINE_DSP_SOURCES}
    ${CLIPLINE_TOP_SOURCES}
    ${CLIPLINE_UI_SOURCES}
    ${CLIPLINE_PRESET_SOURCES})
list(REMOVE_DUPLICATES CLIPLINE_PLUGIN_SOURCES)

# ------------------------------------------------------------------------------------------------
# Embedded resources (fonts, SIL OFL 1.1 - see Resources/Fonts/OFL-Geist.txt)
# ------------------------------------------------------------------------------------------------
juce_add_binary_data(ClipperAssets
    SOURCES
        Resources/Fonts/Geist-Regular.ttf
        Resources/Fonts/Geist-Medium.ttf
        Resources/Fonts/Geist-SemiBold.ttf
        Resources/Fonts/GeistMono-Regular.ttf
        Resources/Fonts/GeistMono-Medium.ttf)
set_target_properties(ClipperAssets PROPERTIES POSITION_INDEPENDENT_CODE ON)

# ------------------------------------------------------------------------------------------------
# The plugin
# ------------------------------------------------------------------------------------------------
set(CLIPLINE_FORMATS VST3)
if(APPLE)
    list(APPEND CLIPLINE_FORMATS AU)          # Logic Pro, GarageBand and FL Studio for Mac use Audio Units
endif()
if(CLIPLINE_BUILD_STANDALONE)
    list(APPEND CLIPLINE_FORMATS Standalone)
endif()

# moduleinfo.json is generated by running the freshly built plugin, which is impossible when cross-compiling
# (e.g. the Windows build made on Linux with cmake/toolchains/llvm-mingw-x86_64.cmake). It is optional for VST3.
if(CMAKE_CROSSCOMPILING)
    set(CLIPLINE_VST3_MANIFEST FALSE)
else()
    set(CLIPLINE_VST3_MANIFEST TRUE)
endif()

juce_add_plugin(Clipline
    VERSION                     ${PROJECT_VERSION}
    COMPANY_NAME                "${CLIPLINE_COMPANY_NAME}"
    BUNDLE_ID                   "${CLIPLINE_BUNDLE_ID}"
    PRODUCT_NAME                "${CLIPLINE_PRODUCT_NAME}"
    PLUGIN_MANUFACTURER_CODE    ${CLIPLINE_MANUFACTURER_CODE}
    PLUGIN_CODE                 ${CLIPLINE_PLUGIN_CODE}
    FORMATS                     ${CLIPLINE_FORMATS}
    IS_SYNTH                    FALSE
    NEEDS_MIDI_INPUT            FALSE
    NEEDS_MIDI_OUTPUT           FALSE
    IS_MIDI_EFFECT              FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS TRUE
    VST3_CATEGORIES             Fx Distortion
    AU_MAIN_TYPE                kAudioUnitType_Effect
    VST3_AUTO_MANIFEST          ${CLIPLINE_VST3_MANIFEST}
    COPY_PLUGIN_AFTER_BUILD     ${CLIPLINE_COPY_AFTER_BUILD})

target_sources(Clipline PRIVATE
    ${CLIPLINE_PLUGIN_SOURCES}
    Source/PluginEntry.cpp)

target_include_directories(Clipline PRIVATE Source)

target_compile_definitions(Clipline PUBLIC
    CLIPLINE_VERSION_STRING="${PROJECT_VERSION}"
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_DISPLAY_SPLASH_SCREEN=0
    JUCE_REPORT_APP_USAGE=0
    JUCE_STRICT_REFCOUNTEDPOINTER=1)

target_link_libraries(Clipline
    PRIVATE
        ClipperAssets
        juce::juce_audio_processors
        juce::juce_dsp
        juce::juce_gui_basics
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_warning_flags)

# MinGW ignores JUCE's #pragma comment(linker, "/MANIFESTDEPENDENCY:...") that requests comctl32 v6. Embed that
# manifest in the plugin DLL ourselves, or real Windows refuses to load it (TaskDialogIndirect missing).
if(MINGW)
    target_sources(Clipline_VST3 PRIVATE cmake/mingw-compat/comctl32-v6.rc)
    set_source_files_properties(cmake/mingw-compat/comctl32-v6.rc PROPERTIES
        OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/cmake/mingw-compat/comctl32-v6.manifest")
endif()

# MSVC: write a PDB next to the plugin (not installed) so crash reports from validators/hosts are readable.
if(MSVC)
    target_compile_options(Clipline PRIVATE /Zi)
    target_link_options(Clipline_VST3 PRIVATE /DEBUG /OPT:REF /OPT:ICF)
endif()

if(CLIPLINE_ENABLE_LTO)
    target_link_libraries(Clipline PUBLIC juce::juce_recommended_lto_flags)
endif()

if(CLIPLINE_BUILD_STANDALONE)
    target_link_libraries(Clipline PRIVATE juce::juce_audio_utils)
endif()

# ------------------------------------------------------------------------------------------------
# Windows installer (single .exe with the plugin embedded)
# ------------------------------------------------------------------------------------------------
option(CLIPLINE_BUILD_INSTALLER "Build the Windows installer (Clipline-<version>-Win64-Setup.exe)" ON)
if(WIN32 AND CLIPLINE_BUILD_INSTALLER)
    add_subdirectory(Installer/Windows)
endif()
# macOS: the .pkg installer is made by Installer/macOS/build-pkg.sh after the build (needs pkgbuild/productbuild).

# ------------------------------------------------------------------------------------------------
# Tests / tools
# ------------------------------------------------------------------------------------------------
if(CLIPLINE_BUILD_TESTS)
    enable_testing()
    add_subdirectory(Tests)
endif()
